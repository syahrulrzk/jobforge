    24→interface EngineState {
    25→  instance: string;
    26→  timer: ReturnType<typeof setInterval> | null;
    27→  ticking: boolean;
    28→  tickCount: number;
    29→  settings: Map<string, string>;
    30→  settingsLoadedAt: number;
    31→  sourceCursor: number;
    32→  engineCursor: number;
    33→  mailtoScanned: Set<string>; // jobIds already scanned for a real mailto contact
    34→}
    35→
    36→const g = globalThis as unknown as { __jobforgeEngine?: EngineState };
    37→
    38→// Unique per module evaluation — lets the engine restart cleanly after HMR
    39→const MODULE_INSTANCE = Math.random().toString(36).slice(2);
    40→
    41→function state(): EngineState {
    42→  if (!g.__jobforgeEngine || g.__jobforgeEngine.instance !== MODULE_INSTANCE) {
    43→    if (g.__jobforgeEngine?.timer) clearInterval(g.__jobforgeEngine.timer);
    44→    g.__jobforgeEngine = {
    45→      instance: MODULE_INSTANCE,
    46→      timer: null,
    47→      ticking: false,
    48→      tickCount: 0,
    49→      settings: new Map(),
    50→      settingsLoadedAt: 0,
    51→      sourceCursor: 0,
    52→      engineCursor: 0,
    53→      mailtoScanned: new Set(),
    54→    };
    55→  }
    56→  return g.__jobforgeEngine;
    57→}

    59→async function getSettings(): Promise<Map<string, string>> {
    60→  const s = state();
    61→  if (Date.now() - s.settingsLoadedAt > 15_000 || s.settings.size === 0) {
    62→    const rows = await db.setting.findMany();
    63→    s.settings = new Map(rows.map((r) => [r.key, r.value]));
    64→    s.settingsLoadedAt = Date.now();
    65→  }
    66→  return s.settings;
    67→}

   125→async function canonicalFor(jobId: string): Promise<CanonicalJob | null> {
   126→  const job = await db.job.findUnique({
   127→    where: { id: jobId },
   128→    include: { company: true, jobLinks: { include: { source: true } }, contact: true },
   129→  });
   130→  if (!job || !job.company || job.jobLinks.length === 0 || !job.contact) return null;
   131→  const link = job.jobLinks[0];
   132→  return {
   133→    source: { platform: link.source.slug, job_id: link.sourceJobId, url: link.sourceUrl },
   134→    company: {
   135→      name: job.company.name,
   136→      logo_url: job.company.logoUrl,
   137→      website: job.company.website ?? null,
   138→      profile: job.company.profile,
   139→    },
   140→    job: {
   141→      title: job.title,
   142→      description: job.description,
   143→      salary:
   144→        job.salaryMin && job.salaryMax
   145→          ? { min: job.salaryMin, max: job.salaryMax, currency: job.currency ?? "IDR" }
   146→          : null,
   147→      location: job.location ?? null,
   148→      employment_type: job.employmentType ?? null,
   149→      workplace_type: job.workplaceType ?? null,
   150→      requirements: job.requirements ? JSON.parse(job.requirements) : null,
   151→      skills: job.skills ? JSON.parse(job.skills) : null,
   152→    },
   153→    contact: {
   154→      hr_email: job.contact.hrEmail,
   155→      email_source: job.contact.emailSourceUrl ?? null,
   156→      email_verified: job.contact.emailVerified,
   157→    },
   158→    metadata: { scraped_at: job.scrapedAt.toISOString() },
   159→  };
   160→}

   162→async function runScrapeForSource(sourceId: string, forced = false): Promise<void> {
   163→  const source = await db.source.findUnique({ where: { id: sourceId } });
   164→  if (!source) return;
   165→  if (!forced && source.status === "INACTIVE") return;
   166→
   167→  const settings = await getSettings();
   168→  const profile = profileFor(source.slug);
   169→
   170→  // §9.2 engine pool — source's pinned engine wins when active, else rotate over the pool
   171→  const pool = parseEnginePool(settings.get(SETTING_KEYS.enginePool));
   172→  const s = state();
   173→  const engine: EngineKey = pool.includes(source.engine as EngineKey)
   174→    ? (source.engine as EngineKey)
   175→    : rotateEngine(pool, s.engineCursor++).engine;
   176→
   177→  // REAL data mode — live public job APIs, throttled for auto runs (be nice to the boards)
   178→  const realMode = (settings.get(SETTING_KEYS.dataMode) ?? "real") === "real";
   179→  const realFetcher = REAL_BOARDS[source.slug];
   180→  if (realMode && realFetcher && !forced) {
   181→    const elapsed = source.lastRunAt ? Date.now() - source.lastRunAt.getTime() : Infinity;
   182→    if (elapsed < 120_000) return;
   183→  }
   184→
   185→  const run = await db.scrapeRun.create({
   186→    data: { sourceId: source.id, engine, status: "RUNNING" },
   187→  });
   188→
   189→  const t0 = Date.now();
   190→  const engineMeta = ENGINES[engine];
   191→  const jitter = engineJitter(engine);
   192→  // engine prep/teardown latency — browser engines boot a driver session (§10.4)
   193→  await new Promise((resolve) => setTimeout(resolve, jitter.latencyMs));
   194→
   195→  let result: { records: RawJobRecord[]; pagesScraped: number; errors: { type: string; message: string }[] };
   196→  if (jitter.failed) {
   197→    // engine-level failure roll (driver crash / renderer hang) — exercises the failover path
   198→    result = { records: [], pagesScraped: 1, errors: [{ type: "NETWORK_ERROR", message: `${engineMeta.name} engine: session crashed mid-run, failover queued` }] };
   199→  } else if (realMode && realFetcher) {
   200→    try {
   201→      result = await realFetcher();
   202→    } catch (err) {
   203→      const e = realBoardError(source.name, err);
   204→      result = { records: [], pagesScraped: 1, errors: [e] };
   205→    }
   206→  } else {
   207→    result = scrapeSource(profile, new Set());
   208→  }
   209→
   210→  let created = 0;
   211→  let updated = 0;
   212→  let duplicated = 0;
   213→
   214→  for (const rec of result.records) {
   215→    const fp = jobFingerprint(rec.rawCompanyName ?? "", rec.rawTitle, rec.rawLocation);
   216→    const existing = await db.job.findUnique({ where: { fingerprint: fp } });
   217→
   218→    if (existing) {
   219→      // §15.1 — duplicate detected: attach source identity, maybe refresh
   220→      const linkExists = await db.jobSource.findFirst({
   221→        where: { jobId: existing.id, sourceId: source.id },
   222→      });
   223→      if (!linkExists) {
   224→        await db.jobSource.create({
   225→          data: {
   226→            jobId: existing.id,
   227→            sourceId: source.id,
   228→            sourceJobId: rec.sourceJobId,
   229→            sourceUrl: rec.sourceUrl,
   230→          },
   231→        });
   232→      }
   233→      if (chance(0.55)) {
   234→        await db.job.update({
   235→          where: { id: existing.id },
   236→          data: { scrapedAt: new Date(rec.scrapedAt) },
   237→        });
   238→        updated += 1;
   239→      } else {
   240→        duplicated += 1;
   241→      }
   242→      continue;
   243→    }
   244→
   245→    const newJob = await db.job.create({
   246→      data: {
   247→        fingerprint: fp,
   248→        title: rec.rawTitle.replace(/\s*[-–]\s*PT\s+.*$/i, "").trim() || rec.rawTitle,
   249→        normalizedTitle: normalizeTitle(rec.rawTitle),
   250→        companyName: rec.rawCompanyName,
   251→        companyLogoUrl: rec.rawCompanyLogoUrl,
   252→        description: rec.rawDescription
   253→          ? rec.rawSalaryText
   254→            ? `${rec.rawDescription}\n\nSalary: ${rec.rawSalaryText}`
   255→            : rec.rawDescription
   256→          : "",
   257→        salaryMin: null, // set at normalize stage
   258→        salaryMax: null,
   259→        currency: null,
   260→        location: rec.rawLocation,
   261→        employmentType: rec.rawEmploymentType,
   262→        workplaceType: rec.rawWorkplaceType,
   263→        requirements: JSON.stringify(rec.rawRequirements),
   264→        skills: JSON.stringify(rec.rawSkills),
   265→        status: "SCRAPED",
   266→        scrapedAt: new Date(rec.scrapedAt),
   267→        companyId: null,
   268→      },
   269→    });
   270→    // §15.2 — store source identity for provenance
   271→    await db.jobSource.create({
   272→      data: {
   273→        jobId: newJob.id,
   274→        sourceId: source.id,
   275→        sourceJobId: rec.sourceJobId,
   276→        sourceUrl: rec.sourceUrl,
   277→      },
   278→    });
   279→    created += 1;
   280→  }
   281→
   282→  const failed = result.errors.length > 0 && result.records.length === 0;
   283→  for (const err of result.errors) {
   284→    await recordError(err.type, err.message, { sourceId: source.id });
   285→    await log("error", "failed", `${err.type}: ${err.message}`, { source: source.slug });
   286→  }
   287→
   288→  await db.scrapeRun.update({
   289→    where: { id: run.id },
   290→    data: {
   291→      finishedAt: new Date(),
   292→      pagesScraped: result.pagesScraped,
   293→      jobsFound: result.records.length,
   294→      jobsCreated: created,
   295→      jobsUpdated: updated,
   296→      jobsDuplicate: duplicated,
   297→      errorCount: result.errors.length,
   298→      status: failed ? "FAILED" : "SUCCESS",
   299→    },
   300→  });
   301→
   302→  await db.source.update({
   303→    where: { id: source.id },
   304→    data: {
   305→      lastRunAt: new Date(),
   306→      status: failed ? "ERROR" : "ACTIVE",
   307→    },
   308→  });
   309→
   310→  await log(
   311→    "scrape",
   312→    failed ? "failed" : "success",
   313→    `Scrape run ${failed ? "failed" : "finished"} via ${engineMeta.name} engine — found ${result.records.length}, created ${created}, updated ${updated}, duplicate ${duplicated}`,
   314→    { source: source.slug, durationMs: Date.now() - t0 }
   315→  );
   316→}

   319→// Phase 2 — Pipeline stages (§14 normalizer, §11 enrichment,
   320→// §12 email discovery, §17 validation)
   321→// ─────────────────────────────────────────────────────────────

   323→
   324→async function advanceStageBatches(): Promise<void> {
   325→  const t = Date.now();
   326→
   327→  // SCRAPED → PROCESSING (normalizer)
   328→  const scraped = await db.job.findMany({
   329→    where: { status: "SCRAPED" },
   330→    take: 24,
   331→    orderBy: { scrapedAt: "asc" },
   332→    include: { jobLinks: { include: { source: true } } },
   333→  });
   334→  for (const job of scraped) {
   335→    // §14 — normalize salary: structured USD/EUR text first, then legacy IDR pattern
   336→    const usd = parseSalaryUsd(job.description);
   337→    const salaryMatch = usd ? null : job.description.match(/IDR\s*([\d.,]+\s*(?:jt|juta)?)/i);
   338→    let salaryMin: number | null = null;
   339→    let salaryMax: number | null = null;
   340→    let currency: string | null = null;
   341→    if (usd) {
   342→      salaryMin = usd.min;
   343→      salaryMax = usd.max;
   344→      currency = usd.currency;
   345→    } else if (salaryMatch) {
   346→      const n = parseFloat(salaryMatch[1].replace(/\./g, "").replace(/,/g, "."));
   347→      const val = /jt|juta/i.test(salaryMatch[1]) ? Math.round(n * 1_000_000) : Math.round(n);
   348→      if (val > 0) {
   349→        salaryMin = val;
   350→        salaryMax = Math.round(val * 1.35);
   351→        currency = "IDR";
   352→      }
   353→    }
   354→    await db.job.update({
   355→      where: { id: job.id },
   356→      data: {
   357→        status: "PROCESSING",
   358→        statusReason: null,
   359→        ...(salaryMin ? { salaryMin, salaryMax, currency } : {}),
   360→      },
   361→    });
   362→  }
   363→  if (scraped.length > 0) {
   364→    await log("normalize", "success", `Normalized ${scraped.length} job record(s) to canonical format`);
   365→  }
   366→
   367→  // PROCESSING → ENRICHING (company enrichment §11)
   368→  const processing = await db.job.findMany({
   369→    where: { status: "PROCESSING" },
   370→    take: 18,
   371→    orderBy: { scrapedAt: "asc" },
   372→  });
   373→  for (const job of processing) {
   374→    const link = await db.jobSource.findFirst({ where: { jobId: job.id }, include: { source: true } });
   375→    const rec = parseRawFromJob(job, link?.sourceUrl ?? "");
   376→    // company upsert by normalized name (§15.3)
   377→    let company = rec.companyName
   378→      ? await db.company.findUnique({ where: { normalizedName: companyFingerprint(rec.companyName) } })
   379→      : null;
   380→    if (!company && rec.companyName) {
   381→      // §11 enrichment — logo: provider-resolved PNG link first, raw source logo as backup
   382→      const resolvedLogo = resolveLogoUrl(rec.companyWebsite) ?? rec.companyLogoUrl ?? "";
   383→      company = await db.company.create({
   384→        data: {
   385→          name: rec.companyName,
   386→          normalizedName: companyFingerprint(rec.companyName),
   387→          logoUrl: resolvedLogo,
   388→          website: rec.companyWebsite,
   389→          profile: rec.companyProfile ?? "Profil perusahaan belum tersedia.",
   390→          industry: null,
   391→        },
   392→      });
   393→      await log("enrich", "info", `New company registered: ${company.name}${resolvedLogo ? " — logo resolved" : ""}`);
   394→    }
   395→    // §11 enrichment — attempt logo/profile recovery from company website
   396→    if (company) {
   397→      const resolvedLogo = resolveLogoUrl(company.website ?? rec.companyWebsite);
   398→      if ((!company.logoUrl || company.logoUrl.endsWith("/assets/logo.png")) && resolvedLogo) {
   399→        await db.company.update({ where: { id: company.id }, data: { logoUrl: resolvedLogo, enrichedAt: new Date() } });
   400→        company = { ...company, logoUrl: resolvedLogo };
   401→        await log("enrich", "success", `Logo resolved via provider chain for ${company.name} (${extractDomain(company.website) ?? "?"})`);
   402→      } else if (!company.logoUrl && rec.companyLogoUrl) {
   403→        await db.company.update({ where: { id: company.id }, data: { logoUrl: rec.companyLogoUrl, enrichedAt: new Date() } });
   404→        company = { ...company, logoUrl: rec.companyLogoUrl };
   405→      }
   406→      if (!company.profile && rec.companyProfile) {
   407→        await db.company.update({ where: { id: company.id }, data: { profile: rec.companyProfile, enrichedAt: new Date() } });
   408→      }
   409→    }
   410→    await db.job.update({
   411→      where: { id: job.id },
   412→      data: { status: "ENRICHING", companyId: company?.id ?? null },
   413→    });
   414→  }
   415→
   416→  // ENRICHING → VALIDATING (HR email discovery §12)
   417→  const enriching = await db.job.findMany({
   418→    where: { status: "ENRICHING" },
   419→    take: 18,
   420→    orderBy: { scrapedAt: "asc" },
   421→    include: { company: true },
   422→  });
   423→  for (const job of enriching) {
   424→    const link = await db.jobSource.findFirst({ where: { jobId: job.id } });
   425→    const rec = parseRawFromJob(job, link?.sourceUrl ?? "");
   426→    if (job.company && rec.publishedEmail) {
   427→      const hasContact = await db.jobContact.findUnique({ where: { jobId: job.id } });
   428→      if (!hasContact) {
   429→        const v = validateEmail(rec.publishedEmail);
   430→        await db.jobContact.create({
   431→          data: {
   432→            jobId: job.id,
   433→            hrEmail: rec.publishedEmail,
   434→            emailSourceUrl: rec.careerPageUrl,
   435→            emailVerified: v.verified,
   436→            emailStatus: v.status,
   437→          },
   438→        });
   439→        if (!v.verified) {
   440→          await recordError(v.reason.split(":")[0] ?? "EMAIL_INVALID", v.reason, { jobId: job.id });
   441→        }
   442→      }
   443→    } else if (job.company) {
   444→      // §13 — email not found: needs enrichment, never guess (§12.4)
   445→      await recordError("EMAIL_NOT_FOUND", `No published recruitment email found on ${rec.careerPageUrl ?? "career page"}`, { jobId: job.id });
   446→    }
   447→    await db.job.update({ where: { id: job.id }, data: { status: "VALIDATING" } });
   448→  }
   449→
   450→  // VALIDATING → READY | NEEDS_ENRICHMENT | REJECTED (§17)
   451→  const validating = await db.job.findMany({
   452→    where: { status: "VALIDATING" },
   453→    take: 18,
   454→    orderBy: { scrapedAt: "asc" },
   455→    include: { company: true, contact: true, jobLinks: { include: { source: true } } },
   456→  });
   457→  for (const job of validating) {
   458→    const res = validateJob({
   459→      companyName: job.company?.name ?? null,
   460→      companyLogoUrl: job.company?.logoUrl || null,
   461→      companyProfile: job.company?.profile || null,
   462→      title: job.title,
   463→      description: job.description,
   464→      hrEmail: job.contact?.hrEmail ?? null,
   465→      emailStatus: (job.contact?.emailStatus as "VALID" | "INVALID" | "UNKNOWN") ?? null,
   466→      sourcePlatform: job.jobLinks[0]?.source.slug ?? null,
   467→      sourceUrl: job.jobLinks[0]?.sourceUrl ?? null,
   468→    });
   469→    await db.job.update({
   470→      where: { id: job.id },
   471→      data: { status: res.outcome, statusReason: res.reason },
   472→    });
   473→    if (res.outcome === "REJECTED") {
   474→      const lastRun = await db.scrapeRun.findFirst({
   475→        where: { sourceId: job.jobLinks[0]?.sourceId ?? "", status: { in: ["SUCCESS", "FAILED"] } },
   476→        orderBy: { startedAt: "desc" },
   477→      });
   478→      if (lastRun) {
   479→        await db.scrapeRun.update({ where: { id: lastRun.id }, data: { jobsRejected: { increment: 1 } } });
   480→      }
   481→      await log("validate", "warning", res.reason, { jobId: job.id });
   482→    } else if (res.outcome === "NEEDS_ENRICHMENT") {
   483→      await log("validate", "warning", res.reason, { jobId: job.id });
   484→    }
   485→  }
   486→
   487→  // NEEDS_ENRICHMENT recovery loop — enrichment worker retry (§36)
   488→  // Real-source jobs get priority (they have genuine enrichment to attempt:
   489→  // a live mailto scan of the posting page); mock jobs follow the simulated path.
   490→  const st = state();
   491→  const needyReal = await db.job.findMany({
   492→    where: { status: "NEEDS_ENRICHMENT", companyName: { not: null }, contact: null },
   493→    take: 4,
   494→    orderBy: { scrapedAt: "asc" },
   495→    include: { company: true, jobLinks: true },
   496→  });
   497→  const needyMock = await db.job.findMany({
   498→    where: { status: "NEEDS_ENRICHMENT", companyName: null },
   499→    take: 4,
   500→    orderBy: { scrapedAt: "asc" },
   501→    include: { company: true, contact: true },
   502→  });
   503→  for (const job of needyReal) {
   504→    const pageUrl = job.jobLinks[0]?.sourceUrl;
   505→    const missingLogo = job.company && !job.company.logoUrl;
   506→    if (missingLogo && job.company) {
   507→      // logo recovery — real payload logo lands with the company at enrich time;
   508→      // for logo-less boards fall back to the job's own payload logo
   509→      const recovered = resolveLogoUrl(job.company.website) ?? job.companyLogoUrl ?? null;
   510→      if (recovered) {
   511→        await db.company.update({
   512→          where: { id: job.company.id },
   513→          data: { logoUrl: recovered, enrichedAt: new Date() },
   514→        });
   515→        await log("enrich", "success", `Logo recovered for ${job.company.name} — re-validating job`, { jobId: job.id });
   516→        await db.job.update({ where: { id: job.id }, data: { status: "VALIDATING" } });
   517→      }
   518→    }
   519→    if (pageUrl && !st.mailtoScanned.has(job.id)) {
   520→      // §12 real enrichment — scan the actual posting page for a published mailto
   521→      // (one attempt per job per process, rate-limited to respect the boards)
   522→      st.mailtoScanned.add(job.id);
   523→      const email = await discoverMailto(pageUrl);
   524→      if (email) {
   525→        const v = validateEmail(email);
   526→        if (v.status !== "INVALID") {
   527→          await db.jobContact.create({
   528→            data: { jobId: job.id, hrEmail: email, emailSourceUrl: pageUrl, emailVerified: v.verified, emailStatus: v.status },
   529→          });
   530→          await log("enrich", "success", `HR email discovered on posting page: ${email}`, { jobId: job.id });
   531→          await db.job.update({ where: { id: job.id }, data: { status: "VALIDATING" } });
   532→        }
   533→      }
   534→    }
   535→  }
   536→  for (const job of needyMock) {
   537→    const missingLogo = job.company && !job.company.logoUrl;
   538→    const missingEmail = !job.contact;
   539→    if (missingLogo && chance(0.4) && job.company) {
   540→      // logo recovery via provider chain (Google → DuckDuckGo) dari domain website
   541→      const recovered = resolveLogoUrl(job.company.website);
   542→      if (recovered) {
   543→        await db.company.update({
   544→          where: { id: job.company.id },
   545→          data: { logoUrl: recovered, enrichedAt: new Date() },
   546→        });
   547→        await log("enrich", "success", `Logo recovered via ${extractDomain(job.company.website) ?? "provider"} for ${job.company.name} — re-validating job`, { jobId: job.id });
   548→        await db.job.update({ where: { id: job.id }, data: { status: "VALIDATING" } });
   549→      }
   550→    }
   551→    if (missingEmail && job.company && job.company.website) {
   552→      // legacy mock sources — template-backed simulated career-page discovery
   553→      if (chance(0.25)) {
   554→        const email = `hr@${job.company.website.replace(/^https?:\/\/(www\.)?/, "")}`;
   555→        const v = validateEmail(email);
   556→        if (v.status === "VALID") {
   557→          await db.jobContact.create({
   558→            data: { jobId: job.id, hrEmail: email, emailSourceUrl: `${job.company.website}/career`, emailVerified: true, emailStatus: "VALID" },
   559→          });
   560→          await log("enrich", "success", `HR email discovered via career page: ${email}`, { jobId: job.id });
   561→          await db.job.update({ where: { id: job.id }, data: { status: "VALIDATING" } });
   562→        }
   563→      }
   564→    }
   565→  }
   566→
   567→  // SENT → PUBLISHED (portal confirms publish async)
   568→  const sentJobs = await db.job.findMany({
   569→    where: { status: "SENT", scrapedAt: { lt: new Date(t - 4_000) } },
   570→    take: 20,
   571→  });
   572→  for (const job of sentJobs) {
   573→    await db.job.update({
   574→      where: { id: job.id },
   575→      data: { status: "PUBLISHED", publishedAt: new Date() },
   576→    });
   577→  }
   578→  if (sentJobs.length > 0) {
   579→    await log("import", "success", `${sentJobs.length} job(s) published to Job Portal`);
   580→  }
   581→}

   585→  job: { title: string; description: string; employmentType: string | null; workplaceType: string | null; companyName?: string | null; companyLogoUrl?: string | null },

   609→    }
   610→    return {
   611→      companyName: job.companyName,

   618→  }

   621→  const m = header.match(/(.+?)\s+sedang mencari/);

   637→/** §14 salary parser for international boards — "USD 180,000 - 190,000", "$25k - $35k", "€60k-80k". */
   638→export function parseSalaryUsd(text: string): { min: number; max: number; currency: string } | null {
   639→  const m = text.match(/(?:USD|[$€])\s?([\d.,]+)\s?([kKmM]?)\s?(?:-|–|—|to)\s?(?:USD\s?)?[$€]?\s?([\d.,]+)\s?([kKmM]?)/);
   640→  if (!m) return null;
   641→  const val = (v: string, u: string): number => {
   642→    const n = parseFloat(v.replace(/,/g, ""));
   643→    if (!Number.isFinite(n)) return 0;
   644→    if (u === "k" || u === "K") return Math.round(n * 1_000);
   645→    if (u === "m" || u === "M") return Math.round(n * 1_000_000);
   646→    return Math.round(n);
   647→  };
   648→  const min = val(m[1], m[2]);
   649→  const max = val(m[3], m[4]);
   650→  if (min > 0 && max >= min) return { min, max, currency: m[0].includes("€") ? "EUR" : "USD" };
   651→  return null;
   652→}

   654→/** §12 real HR email discovery — fetch the posting page and look for a published contact. */
   655→async function discoverMailto(pageUrl: string): Promise<string | null> {
   656→  try {
   657→    const res = await fetch(pageUrl, {
   658→      signal: AbortSignal.timeout(6_000),
   659→      redirect: "follow",
   660→      headers: { "User-Agent": "JobForgeBot/1.0 (+HR email discovery §12)", Accept: "text/html" },
   661→      cache: "no-store",
   662→    });
   663→    if (!res.ok) return null;
   664→    const html = (await res.text()).slice(0, 400_000);
   665→    const mailto = html.match(/mailto:([A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,})/i);
   666→    if (mailto) return mailto[1].toLowerCase();
   667→    const named = html.match(/\b((?:hr|careers?|recruitment|talent|jobs?|karir)[A-Za-z0-9._%+-]*@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)+)\b/i);
   668→    return named ? named[1].toLowerCase() : null;
   669→  } catch {
   670→    return null; // page unreachable — stays NEEDS_ENRICHMENT, honest per §12.4
   671→  }
   672→}

   682→async function deliverReadyJobs(settings: Map<string, string>): Promise<void> {
   683→  const autoDelivery = settings.get(SETTING_KEYS.autoDelivery) !== "false";
   684→  if (!autoDelivery) return;
   685→  const batchSize = Math.max(1, parseInt(settings.get(SETTING_KEYS.batchSize) ?? "25", 10) || 25);
   686→  const maxAttempts = Math.max(1, parseInt(settings.get(SETTING_KEYS.maxAttempts) ?? "3", 10) || 3);
   687→
   688→  // 1. advance existing SENDING/PENDING deliveries (incl. retries §21)
   689→  const inFlight = await db.apiDelivery.findMany({
   690→    where: { status: { in: ["PENDING", "SENDING"] } },
   691→    take: 20,
   692→    include: { job: true },
   693→  });
   694→  for (const d of inFlight) {
   695→    if (d.nextRetryAt && d.nextRetryAt.getTime() > Date.now()) continue;
   696→    await db.apiDelivery.update({ where: { id: d.id }, data: { status: "SENDING" } });
   697→    const canonical = await canonicalFor(d.jobId);
   698→    // simulate transient API failure (§21 retry strategy)
   699→    const shouldFail = chance(0.07) && d.attempt < maxAttempts;
   700→    if (!canonical || shouldFail) {
   701→      const failedFinal = d.attempt >= maxAttempts;
   702→      const backoffMs = [5_000, 15_000, 45_000][Math.min(d.attempt - 1, 2)];
   703→      await db.apiDelivery.update({
   704→        where: { id: d.id },
   705→        data: {
   706→          status: failedFinal ? "FAILED" : "PENDING",
   707→          attempt: failedFinal ? d.attempt : d.attempt + 1,
   708→          responseCode: shouldFail ? 500 : null,
   709→          responseBody: shouldFail ? JSON.stringify({ success: false, message: "Internal Server Error" }) : null,
   710→          nextRetryAt: failedFinal ? null : new Date(Date.now() + backoffMs),
   711→        },
   712→      });
   713→      if (failedFinal) {
   714→        await recordError("API_ERROR", `Portal import failed after ${maxAttempts} attempts`, { jobId: d.jobId });
   715→        await log("deliver", "failed", `Delivery ${d.requestId} failed permanently after ${maxAttempts} attempts`, { jobId: d.jobId });
   716→      } else {
   717→        await log("deliver", "warning", `Delivery ${d.requestId} attempt ${d.attempt} failed (500) — retry in ${backoffMs / 1000}s`, { jobId: d.jobId });
   718→      }
   719→      continue;
   720→    }
   721→
   722→    const payload = { source: canonical.source.platform, scraped_at: canonical.metadata.scraped_at, jobs: [canonical] };
   723→    const result = await processBulkImport(payload);
   724→    await db.apiDelivery.update({
   725→      where: { id: d.id },
   726→      data: {
   727→        status: result.success ? "SUCCESS" : "FAILED",
   728→        responseCode: result.success ? 200 : 422,
   729→        responseBody: JSON.stringify(result),
   730→        deliveredAt: result.success ? new Date() : null,
   731→        nextRetryAt: null,
   732→      },
   733→    });
   734→    if (result.success) {
   735→      await db.job.update({ where: { id: d.jobId }, data: { status: "SENT", statusReason: null } });
   736→      await log("deliver", "success", `Import accepted — ${result.data.created} created, ${result.data.updated} updated`, { jobId: d.jobId });
   737→    }
   738→  }
   739→
   740→  // 2. create new deliveries for READY jobs without one
   741→  const readyJobs = await db.job.findMany({
   742→    where: { status: "READY", deliveries: { none: { status: { in: ["PENDING", "SENDING"] } } } },
   743→    take: batchSize,
   744→    orderBy: { scrapedAt: "asc" },
   745→    include: { deliveries: { orderBy: { createdAt: "desc" }, take: 1 } },
   746→  });
   747→  for (const job of readyJobs) {
   748→    const last = job.deliveries[0];
   749→    if (last && last.status === "FAILED" && last.attempt >= maxAttempts) continue; // needs manual retry
   750→    await db.apiDelivery.create({
   751→      data: {
   752→        jobId: job.id,
   753→        endpoint: DELIVERY_ENDPOINT,
   754→        requestId: generateRequestId(),
   755→        attempt: 1,
   756→        maxAttempts,
   757→        status: "PENDING",
   758→      },
   759→    });
   760→  }
   761→}

   767→async function tick(): Promise<void> {
   768→  const s = state();
   769→  if (s.ticking) return;
   770→  s.ticking = true;
   771→  try {
   772→    const settings = await getSettings();
   773→    s.tickCount += 1;
   774→
   775→    // scrape phase
   776→    const autoScrape = settings.get(SETTING_KEYS.autoScrape) !== "false";
   777→    if (autoScrape && s.tickCount % 2 === 1) {
   778→      const cap = parseInt(settings.get(SETTING_KEYS.demoJobCap) ?? "800", 10) || 800;
   779→      const total = await db.job.count();
   780→      if (total < cap) {
   781→        const sources = await db.source.findMany({ where: { status: "ACTIVE" }, orderBy: { id: "asc" } });
   782→        if (sources.length > 0) {
   783→          s.sourceCursor = (s.sourceCursor + 1) % sources.length;
   784→          await runScrapeForSource(sources[s.sourceCursor].id);
   785→        }
   786→      }
   787→    }
   788→
   789→    await advanceStageBatches();
   790→    await deliverReadyJobs(settings);
   791→
   792→    // housekeeping: keep activity log bounded
   793→    if (s.tickCount % 40 === 0) {
   794→      const old = await db.activityLog.findMany({ orderBy: { ts: "desc" }, skip: 300, take: 1000, select: { id: true } });
   795→      if (old.length > 0) {
   796→        await db.activityLog.deleteMany({ where: { id: { in: old.map((o) => o.id) } } });
   797→      }
   798→    }
   799→  } catch (err) {
   800→    await recordError("DATABASE_ERROR", `Engine tick error: ${err instanceof Error ? err.message : String(err)}`);
   801→  } finally {
   802→    s.ticking = false;
   803→  }
   804→}

   806→export function startEngine(): void {
   807→  const s = state();
   808→  if (s.timer) return;
   809→  getSettings()
   810→    .then((settings) => {
   811→      const interval = Math.max(2000, parseInt(settings.get(SETTING_KEYS.tickIntervalMs) ?? "5000", 10) || 5000);
   812→      s.timer = setInterval(() => void tick(), interval);
   813→      if (typeof s.timer === "object" && "unref" in s.timer) (s.timer as { unref: () => void }).unref();
   814→      void log("scrape", "info", `Worker engine started — tick every ${interval / 1000}s`);
   815→    })
   816→    .catch(() => {
   817→      // db not ready yet — retry shortly
   818→      setTimeout(() => {
   819→        s.timer = null;
   820→        startEngine();
   821→      }, 3000);
   822→    });
   823→}
