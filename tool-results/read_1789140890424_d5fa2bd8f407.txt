     1→"use client";
     2→
     3→// Cari LokerBase (PRD §29 search) — dua mode pencarian:
     4→//
     5→//   1. Cari Database  — instan, filter job yang sudah ada di DB
     6→//   2. Scrape Live    — pilih board dari Data Sources, engine jalan
     7→//                       scraping on-demand untuk kata kunci posisi;
     8→//                       engine tiap board ngikutin setting source
     9→//                       (§9.3 multi-engine), hasil masuk DB
    10→//                       (fingerprint dedup) lalu ditampilkan dari DB
    11→//
    12→// Aturan email wajib: hanya lowongan yang punya email HR yang masuk DB
    13→// dan tampil di hasil (withEmail=1) — sisanya dilewati biar tidak spam.
    14→//
    15→// Hasil kartu membuka detail lengkap via JobDetailSheet.
    16→
    17→import { useEffect, useMemo, useState } from "react";
    18→import { toast } from "sonner";
    19→import { Badge } from "@/components/ui/badge";
    20→import { Button } from "@/components/ui/button";
    21→import { Input } from "@/components/ui/input";
    22→import { Switch } from "@/components/ui/switch";
    23→import { AlertTriangle, BriefcaseBusiness, ChevronLeft, ChevronRight, Database, Globe, Layers, Loader2, MapPin, Search, Zap } from "lucide-react";
    24→import { CompanyAvatar, EmptyState, StatusBadge, formatIDR, timeAgo } from "./ui-bits";
    25→import { JobDetailSheet } from "./job-detail-sheet";
    26→import { ENGINES, type EngineKey } from "@/lib/jobforge/engines";
    27→import { useApi } from "@/hooks/use-api";
    28→
    29→interface JobRow {
    30→  id: string;
    31→  title: string;
    32→  company: { id: string; name: string; logoUrl: string; website: string | null } | null;
    33→  source: { slug: string; name: string; url: string } | null;
    34→  location: string | null;
    35→  workplaceType: string | null;
    36→  salaryMin: number | null;
    37→  salaryMax: number | null;
    38→  currency: string | null;
    39→  status: string;
    40→  scrapedAt: string;
    41→}
    42→
    43→interface JobsResponse {
    44→  total: number;
    45→  page: number;
    46→  pageSize: number;
    47→  jobs: JobRow[];
    48→}
    49→
    50→interface LiveEngineAttempt {
    51→  engine: EngineKey;
    52→  status: "success" | "failed";
    53→  httpStatus?: number;
    54→  durationMs: number;
    55→  note?: string;
    56→}
    57→
    58→interface LiveBoardResult {
    59→  board: string;
    60→  slug: string;
    61→  engine: EngineKey;
    62→  engines: EngineKey[];
    63→  attempts: LiveEngineAttempt[];
    64→  status: "success" | "failed";
    65→  found: number;
    66→  created: number;
    67→  duplicate: number;
    68→  skipped: number;
    69→  durationMs: number;
    70→  error?: string;
    71→}
    72→
    73→interface LiveSearchResponse {
    74→  q: string;
    75→  durationMs: number;
    76→  boards: LiveBoardResult[];
    77→  totalFound: number;
    78→  totalCreated: number;
    79→  totalDuplicate: number;
    80→  totalSkipped: number;
    81→}
    82→
    83→// Source dari Data Sources — dipakai sebagai picker board di mode live
    84→interface SourceRow {
    85→  slug: string;
    86→  name: string;
    87→  status: string;
    88→  engine: string;
    89→  engines: string;
    90→}
    91→
    92→const STATUS_DOT: Record<string, string> = {
    93→  ACTIVE: "bg-emerald-400",
    94→  ERROR: "bg-rose-400",
    95→  INACTIVE: "bg-zinc-600",
    96→};
    97→
    98→function primaryEngine(s: SourceRow): string {
    99→  return (s.engines || s.engine || "cheerio").split(",")[0].trim();
   100→}
   101→
   102→function engineChainLabel(s: SourceRow): string {
   103→  return (s.engines || s.engine || "cheerio")
   104→    .split(",")
   105→    .map((e) => ENGINES[e.trim() as EngineKey]?.name ?? e.trim())
   106→    .join(" → ");
   107→}
   108→
   109→const POPULAR_POSITIONS = [
   110→  "Frontend Developer",
   111→  "Backend Developer",
   112→  "Fullstack Developer",
   113→  "Data Analyst",
   114→  "DevOps Engineer",
   115→  "UI/UX Designer",
   116→  "Product Manager",
   117→  "Business Analyst",
   118→];
   119→
   120→type Mode = "db" | "live";
   121→
   122→interface Applied {
   123→  q: string;
   124→  loc: string;
   125→  remote: boolean;
   126→}
   127→
   128→export function SearchView() {
   129→  const [mode, setMode] = useState<Mode>("db");
   130→  const [input, setInput] = useState("");
   131→  const [locInput, setLocInput] = useState("");
   132→  const [remote, setRemote] = useState(false);
   133→  const [applied, setApplied] = useState<Applied>({ q: "", loc: "", remote: false });
   134→  const [page, setPage] = useState(1);
   135→  const [selectedId, setSelectedId] = useState<string | null>(null);
   136→
   137→  const [liveRunning, setLiveRunning] = useState(false);
   138→  const [liveResult, setLiveResult] = useState<LiveSearchResponse | null>(null);
   139→
   140→  // Board picker — daftar source dari Data Sources (DB), bukan hardcode.
   141→  // Default terpilih: semua source ACTIVE.
   142→  const [sources, setSources] = useState<SourceRow[]>([]);
   143→  const [selectedBoards, setSelectedBoards] = useState<Set<string>>(new Set());
   144→
   145→  useEffect(() => {
   146→    void (async () => {
   147→      try {
   148→        const res = await fetch("/api/sources", { cache: "no-store" });
   149→        if (!res.ok) return;
   150→        const json = (await res.json()) as { sources?: SourceRow[] };
   151→        const rows = json.sources ?? [];
   152→        setSources(rows);
   153→        setSelectedBoards(new Set(rows.filter((s) => s.status === "ACTIVE").map((s) => s.slug)));
   154→      } catch {
   155→        // picker tetap kosong — mode live pakai default semua ACTIVE
   156→      }
   157→    })();
   158→  }, []);
   159→
   160→  const toggleBoard = (slug: string) => {
   161→    setSelectedBoards((prev) => {
   162→      const next = new Set(prev);
   163→      if (next.has(slug)) next.delete(slug);
   164→      else next.add(slug);
   165→      return next;
   166→    });
   167→  };
   168→
   169→  const selectedNames = useMemo(
   170→    () => sources.filter((s) => selectedBoards.has(s.slug)).map((s) => s.name),
   171→    [sources, selectedBoards]
   172→  );
   173→
   174→  const applyNow = (q = input, loc = locInput, rem = remote) => {
   175→    setApplied({ q: q.trim(), loc: loc.trim(), remote: rem });
   176→    setPage(1);
   177→  };
   178→
   179→  // debounce 350ms — instant search as you type (database mode only)
   180→  useEffect(() => {
   181→    if (mode !== "db") return;
   182→    const t = setTimeout(() => setApplied({ q: input.trim(), loc: locInput.trim(), remote }), 350);
   183→    return () => clearTimeout(t);
   184→  }, [input, locInput, remote, mode]);
   185→
   186→  const url = useMemo(() => {
   187→    const sp = new URLSearchParams({ page: String(page), pageSize: "18", withEmail: "1" });
   188→    if (applied.q) sp.set("title", applied.q);
   189→    if (applied.loc) sp.set("location", applied.loc);
   190→    if (applied.remote) sp.set("remote", "true");
   191→    return `/api/jobs?${sp.toString()}`;
   192→  }, [applied, page]);
   193→
   194→  const { data, loading } = useApi<JobsResponse>(url);
   195→  const totalPages = data ? Math.max(1, Math.ceil(data.total / data.pageSize)) : 1;
   196→  const hasQuery = applied.q || applied.loc || applied.remote;
   197→
   198→  // ── Live scrape flow (mode "live") ──
   199→  const runLiveSearch = async (q = input) => {
   200→    const keyword = q.trim();
   201→    if (keyword.length < 2) {
   202→      toast.error("Ketik posisi yang mau dicari (min. 2 karakter)");
   203→      return;
   204→    }
   205→    if (sources.length > 0 && selectedBoards.size === 0) {
   206→      toast.error("Pilih minimal 1 board sumber dulu");
   207→      return;
   208→    }
   209→    setLiveRunning(true);
   210→    setLiveResult(null);
   211→    toast.info(`Engine menjalankan scrape live ke ${selectedBoards.size} board untuk “${keyword}”…`);
   212→    try {
   213→      const res = await fetch("/api/search/live", {
   214→        method: "POST",
   215→        headers: { "Content-Type": "application/json" },
   216→        body: JSON.stringify({ q: keyword, sources: [...selectedBoards] }),
   217→      });
   218→      const json = (await res.json()) as LiveSearchResponse & { error?: string };
   219→      if (!res.ok) throw new Error(json.error ?? "Live scrape gagal");
   220→      setLiveResult(json);
   221→      applyNow(keyword, locInput, remote);
   222→      if (json.totalCreated > 0) {
   223→        toast.success(`Scrape selesai ${(json.durationMs / 1000).toFixed(1)}s — ${json.totalCreated} job baru tersimpan${json.totalSkipped > 0 ? `, ${json.totalSkipped} dilewati (tanpa email HR)` : ""}`);
   224→      } else {
   225→        toast.info(`Scrape selesai — ${json.totalFound} hasil, semua sudah ada di database`);
   226→      }
   227→    } catch (e) {
   228→      toast.error(e instanceof Error ? e.message : "Live scrape gagal");
   229→    } finally {
   230→      setLiveRunning(false);
   231→    }
   232→  };
   233→
   234→  const pickPosition = (p: string) => {
   235→    setInput(p);
   236→    if (mode === "db") applyNow(p, locInput, remote);
   237→    else void runLiveSearch(p);
   238→  };
   239→
   240→  const switchMode = (m: Mode) => {
   241→    setMode(m);
   242→    if (m === "db") applyNow(); // instant db search with current keywords
   243→  };
   244→
   245→  return (
   246→    <div className="space-y-4">
   247→      {/* Hero search */}
   248→      <section className="rounded-xl border border-zinc-800 bg-gradient-to-br from-zinc-900 via-zinc-900/60 to-zinc-900/20 p-6">
   249→        <div className="flex flex-wrap items-center justify-between gap-3">
   250→          <div className="flex items-center gap-2.5">
   251→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-amber-500/15">
   252→              <BriefcaseBusiness className="h-4.5 w-4.5 text-amber-400" />
   253→            </div>
   254→            <div>
   255→              <h2 className="text-base font-bold text-zinc-100">Cari LokerBase</h2>
   256→              <p className="text-xs text-zinc-500">
   257→                {mode === "db"
   258→                  ? "Filter instan dari database job hasil scraping pipeline"
   259→                  : "Pilih board dari Data Sources — engine scrape ngikutin setting engine di tiap source"}
   260→              </p>
   261→            </div>
   262→          </div>
   263→
   264→          {/* Mode toggle */}
   265→          <div className="inline-flex rounded-lg border border-zinc-800 bg-zinc-950 p-0.5">
   266→            <button
   267→              onClick={() => switchMode("db")}
   268→              className={`inline-flex items-center gap-1.5 rounded-md px-3 py-1.5 text-xs font-medium transition-colors ${
   269→                mode === "db" ? "bg-amber-500/15 text-amber-300" : "text-zinc-500 hover:text-zinc-300"
   270→              }`}
   271→            >
   272→              <Database className="h-3.5 w-3.5" /> Cari Database
   273→            </button>
   274→            <button
   275→              onClick={() => switchMode("live")}
   276→              className={`inline-flex items-center gap-1.5 rounded-md px-3 py-1.5 text-xs font-medium transition-colors ${
   277→                mode === "live" ? "bg-teal-500/15 text-teal-300" : "text-zinc-500 hover:text-zinc-300"
   278→              }`}
   279→            >
   280→              <Zap className="h-3.5 w-3.5" /> Scrape Live (Engine)
   281→            </button>
   282→          </div>
   283→        </div>
   284→
   285→        <div className="mt-4 flex flex-col gap-2 sm:flex-row">
   286→          <div className="relative min-w-0 flex-1">
   287→            <Search className="absolute top-1/2 left-3 h-4 w-4 -translate-y-1/2 text-zinc-500" />
   288→            <Input
   289→              autoFocus
   290→              placeholder={mode === "db" ? "Posisi yang dicari — mis. frontend developer, data analyst…" : "Posisi buat di-scrape live — mis. business analyst…"}
   291→              value={input}
   292→              onChange={(e) => setInput(e.target.value)}
   293→              onKeyDown={(e) => e.key === "Enter" && (mode === "db" ? applyNow() : void runLiveSearch())}
   294→              disabled={liveRunning}
   295→              className="h-11 border-zinc-800 bg-zinc-950 pl-9 text-sm"
   296→            />
   297→          </div>
   298→          <div className="relative sm:w-56">
   299→            <MapPin className="absolute top-1/2 left-3 h-4 w-4 -translate-y-1/2 text-zinc-500" />
   300→            <Input
   301→              placeholder="Lokasi (opsional)"
   302→              value={locInput}
   303→              onChange={(e) => setLocInput(e.target.value)}
   304→              onKeyDown={(e) => e.key === "Enter" && (mode === "db" ? applyNow() : void runLiveSearch())}
   305→              disabled={liveRunning}
   306→              className="h-11 border-zinc-800 bg-zinc-950 pl-9 text-sm"
   307→            />
   308→          </div>
   309→          {mode === "db" ? (
   310→            <Button className="h-11 bg-amber-500 px-5 text-zinc-950 hover:bg-amber-400" onClick={() => applyNow()}>
   311→              <Search className="h-4 w-4" /> Cari
   312→            </Button>
   313→          ) : (
   314→            <Button
   315→              className="h-11 bg-teal-500 px-5 text-zinc-950 hover:bg-teal-400"
   316→              onClick={() => void runLiveSearch()}
   317→              disabled={liveRunning}
   318→            >
   319→              {liveRunning ? <Loader2 className="h-4 w-4 animate-spin" /> : <Zap className="h-4 w-4" />}
   320→              {liveRunning ? "Scraping…" : "Scrape Sekarang"}
   321→            </Button>
   322→          )}
   323→        </div>
   324→
   325→        {/* Board picker (live mode) — sumber = Data Sources yang sudah ditambahkan */}
   326→        {mode === "live" && (
   327→          <div className="mt-3 flex flex-wrap items-center gap-1.5">
   328→            <span className="mr-1 flex items-center gap-1 text-[11px] text-zinc-500">
   329→              <Layers className="h-3.5 w-3.5" /> Board sumber:
   330→            </span>
   331→            {sources.map((s) => {
   332→              const on = selectedBoards.has(s.slug);
   333→              const pe = ENGINES[primaryEngine(s) as EngineKey];
   334→              return (
   335→                <button
   336→                  key={s.slug}
   337→                  type="button"
   338→                  disabled={liveRunning}
   339→                  onClick={() => toggleBoard(s.slug)}
   340→                  title={`${s.name} · ${s.status} · engine: ${engineChainLabel(s)}`}
   341→                  className={`inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-[11px] transition-colors disabled:opacity-50 ${
   342→                    on
   343→                      ? "border-teal-500/50 bg-teal-500/15 text-teal-200"
   344→                      : "border-zinc-800 bg-zinc-900 text-zinc-500 hover:border-zinc-700 hover:text-zinc-300"
   345→                  }`}
   346→                >
   347→                  <span className={`h-1.5 w-1.5 rounded-full ${STATUS_DOT[s.status] ?? "bg-zinc-600"}`} />
   348→                  {s.name}
   349→                  <span className={`rounded px-1 text-[9px] font-semibold ${on && pe ? pe.badge : "bg-zinc-800 text-zinc-500"}`}>
   350→                    {pe?.name ?? primaryEngine(s)}
   351→                  </span>
   352→                </button>
   353→              );
   354→            })}
   355→            {sources.length === 0 && (
   356→              <span className="text-[11px] text-zinc-600">
   357→                Belum ada source — tambahkan dulu di menu Data Sources
   358→              </span>
   359→            )}
   360→            {sources.length > 0 && selectedBoards.size < sources.length && (
   361→              <button
   362→                type="button"
   363→                disabled={liveRunning}
   364→                onClick={() => setSelectedBoards(new Set(sources.filter((s) => s.status === "ACTIVE").map((s) => s.slug)))}
   365→                className="rounded-full border border-dashed border-zinc-700 px-2.5 py-1 text-[11px] text-zinc-500 transition-colors hover:border-zinc-600 hover:text-zinc-300 disabled:opacity-50"
   366→              >
   367→                Reset ke semua aktif
   368→              </button>
   369→            )}
   370→          </div>
   371→        )}
   372→
   373→        <div className="mt-3 flex flex-wrap items-center gap-1.5">
   374→          <span className="mr-1 text-[11px] text-zinc-600">{mode === "db" ? "Populer:" : "Coba scrape:"}</span>
   375→          {POPULAR_POSITIONS.map((p) => (
   376→            <button
   377→              key={p}
   378→              disabled={liveRunning}
   379→              onClick={() => pickPosition(p)}
   380→              className={`rounded-full border px-2.5 py-1 text-[11px] transition-colors disabled:opacity-50 ${
   381→                applied.q.toLowerCase() === p.toLowerCase()
   382→                  ? mode === "live"
   383→                    ? "border-teal-500/50 bg-teal-500/15 text-teal-300"
   384→                    : "border-amber-500/50 bg-amber-500/15 text-amber-300"
   385→                  : "border-zinc-800 bg-zinc-900 text-zinc-400 hover:border-zinc-700 hover:text-zinc-200"
   386→              }`}
   387→            >
   388→              {p}
   389→            </button>
   390→          ))}
   391→          <label className="ml-auto flex cursor-pointer items-center gap-2 text-[11px] text-zinc-400">
   392→            <Globe className="h-3.5 w-3.5 text-teal-400" /> Hanya remote
   393→            <Switch checked={remote} onCheckedChange={(v) => setRemote(v)} className="data-[state=checked]:bg-teal-500" />
   394→          </label>
   395→        </div>
   396→
   397→        {mode === "live" && liveRunning && (
   398→          <p className="mt-3 flex items-center gap-2 rounded-lg border border-teal-500/20 bg-teal-500/5 px-3 py-2 text-xs text-teal-300">
   399→            <Loader2 className="h-3.5 w-3.5 animate-spin" />
   400→            Engine menjalankan scrape live ke {selectedNames.length > 0 ? selectedNames.join(", ") : "board aktif"} — engine ngikutin setting tiap source · hanya lowongan dengan email HR yang disimpan…
   401→          </p>
   402→        )}
   403→
   404→        {/* Live scrape breakdown per board */}
   405→        {mode === "live" && liveResult && (
   406→          <div className="mt-3 space-y-2">
   407→            <p className="text-xs text-zinc-500">
   408→              Selesai dalam <span className="font-semibold tabular-nums text-zinc-300">{(liveResult.durationMs / 1000).toFixed(1)}s</span> —{" "}
   409→              <span className="font-semibold text-teal-300">{liveResult.totalCreated} job baru</span> tersimpan,{" "}
   410→              {liveResult.totalDuplicate} duplikat dilewati{" "}
   411→              {liveResult.totalSkipped > 0 && (
   412→                <>
   413→                  · <span className="text-zinc-400">{liveResult.totalSkipped} dilewati (tanpa email HR)</span>{" "}
   414→                </>
   415→              )}
   416→              · {liveResult.totalFound} hasil ditemukan
   417→            </p>
   418→            <div className="flex flex-wrap gap-1.5">
   419→              {liveResult.boards.map((b) => {
   420→                const attemptDetail = (b.attempts ?? []).length
   421→                  ? (b.attempts as LiveEngineAttempt[])
   422→                      .map((a) => `${ENGINES[a.engine as EngineKey]?.name ?? a.engine}: ${a.status === "success" ? "sukses" : "gagal"}${a.httpStatus ? ` HTTP ${a.httpStatus}` : ""} (${a.durationMs}ms)${a.note ? ` — ${a.note}` : ""}`)
   423→                      .join("  |  ")
   424→                  : null;
   425→                return (
   426→                  <span
   427→                    key={b.slug}
   428→                    className={`inline-flex items-center gap-1.5 rounded-md border px-2 py-1 text-[11px] ${
   429→                      b.status === "success"
   430→                        ? "border-zinc-800 bg-zinc-900 text-zinc-300"
   431→                        : "border-rose-500/30 bg-rose-500/5 text-rose-300"
   432→                    }`}
   433→                    title={attemptDetail ?? b.error ?? `${b.board} — ${b.durationMs}ms`}
   434→                  >
   435→                    <span className={`h-1.5 w-1.5 rounded-full ${b.status === "success" ? "bg-emerald-400" : "bg-rose-400"}`} />
   436→                    {b.board}
   437→                    {/* seluruh engine yang dicoba — failover chain terlihat, bukan cuma 1 */}
   438→                    {(b.attempts ?? [{ engine: b.engine, status: b.status, durationMs: b.durationMs }]).map((a, ai) => {
   439→                      const m = ENGINES[a.engine as EngineKey];
   440→                      const fail = a.status !== "success";
   441→                      return (
   442→                        <span
   443→                          key={`${b.slug}-${ai}`}
   444→                          className={`rounded px-1 text-[9px] font-semibold ${
   445→                            fail ? "bg-zinc-800 text-zinc-500 line-through decoration-zinc-600" : m?.badge ?? "bg-zinc-800 text-zinc-400"
   446→                          }`}
   447→                        >
   448→                          {m?.name ?? a.engine}
   449→                        </span>
   450→                      );
   451→                    })}
   452→                    {b.status === "success" ? (
   453→                      <span className="tabular-nums">
   454→                        <span className="font-semibold text-teal-300">{b.created}</span> baru / {b.found} found
   455→                        {b.skipped > 0 && <span className="text-zinc-500"> · {b.skipped} no-email</span>}
   456→                      </span>
   457→                    ) : (
   458→                      <span>
   459→                        gagal
   460→                        {(b.attempts?.length ?? 0) > 1 && <span className="text-zinc-500"> · {b.attempts.length} engine</span>}
   461→                      </span>
   462→                    )}
   463→                  </span>
   464→                );
   465→              })}
   466→            </div>
   467→            {/* Alasan gagal ditulis LANGSUNG di sini — bukan cuma di tooltip */}
   468→            {liveResult.boards.some((b) => b.status === "failed" && b.error) && (
   469→              <div className="space-y-1.5 rounded-lg border border-rose-500/20 bg-rose-500/5 px-3 py-2">
   470→                {liveResult.boards
   471→                  .filter((b) => b.status === "failed" && b.error)
   472→                  .map((b) => (
   473→                    <p key={b.slug} className="flex items-start gap-1.5 text-[11px] leading-relaxed text-rose-300">
   474→                      <AlertTriangle className="mt-0.5 h-3 w-3 shrink-0" />
   475→                      <span>
   476→                        <span className="font-semibold">{b.board}</span> — {b.error}
   477→                      </span>
   478→                    </p>
   479→                  ))}

   598→
   599→      {/* Pagination */}
