   127→   127→
   128→   128→export function SearchView() {
   129→   129→  const [mode, setMode] = useState<Mode>("db");
   130→   130→  const [input, setInput] = useState("");
   131→   131→  const [locInput, setLocInput] = useState("");
   132→   132→  const [remote, setRemote] = useState(false);
   133→   133→  const [applied, setApplied] = useState<Applied>({ q: "", loc: "", remote: false });
   134→   134→  const [page, setPage] = useState(1);
   135→   135→  const [selectedId, setSelectedId] = useState<string | null>(null);
   136→   136→
   137→   137→  const [liveRunning, setLiveRunning] = useState(false);
   138→   138→  const [liveResult, setLiveResult] = useState<LiveSearchResponse | null>(null);
   139→   139→
   140→   140→  // Board picker — daftar source dari Data Sources (DB), bukan hardcode.
   141→   141→  // Default terpilih: semua source ACTIVE.
   142→   142→  const [sources, setSources] = useState<SourceRow[]>([]);
   143→   143→  const [selectedBoards, setSelectedBoards] = useState<Set<string>>(new Set());
   144→   144→
   145→   145→  useEffect(() => {
   146→   146→    void (async () => {
   147→   147→      try {
   148→   148→        const res = await fetch("/api/sources", { cache: "no-store" });
   149→   149→        if (!res.ok) return;
   150→   150→        const json = (await res.json()) as { sources?: SourceRow[] };
   151→   151→        const rows = json.sources ?? [];
   152→   152→        setSources(rows);
   153→   153→        setSelectedBoards(new Set(rows.filter((s) => s.status === "ACTIVE").map((s) => s.slug)));
   154→   154→      } catch {
   155→   155→        // picker tetap kosong — mode live pakai default semua ACTIVE
   156→   156→      }
   157→   157→    })();
   158→   158→  }, []);
   159→   159→
   160→   160→  const toggleBoard = (slug: string) => {
   161→   161→    setSelectedBoards((prev) => {
   162→   162→      const next = new Set(prev);
   163→   163→      if (next.has(slug)) next.delete(slug);
   164→   164→      else next.add(slug);
   165→   165→      return next;
   166→   166→    });
   167→   167→  };
   168→   168→
   169→   169→  const selectedNames = useMemo(
   170→   170→    () => sources.filter((s) => selectedBoards.has(s.slug)).map((s) => s.name),
   171→   171→    [sources, selectedBoards]
   172→   172→  );
   173→   173→
   174→   174→  const applyNow = (q = input, loc = locInput, rem = remote) => {
   175→   175→    setApplied({ q: q.trim(), loc: loc.trim(), remote: rem });
   176→   176→    setPage(1);
   177→   177→  };
   178→   178→
   179→   179→  // debounce 350ms — instant search as you type (database mode only)
   180→   180→  useEffect(() => {
   181→   181→    if (mode !== "db") return;
   182→   182→    const t = setTimeout(() => setApplied({ q: input.trim(), loc: locInput.trim(), remote }), 350);
   183→   183→    return () => clearTimeout(t);
   184→   184→  }, [input, locInput, remote, mode]);
   185→   185→
   186→   186→  const url = useMemo(() => {
   187→   187→    const sp = new URLSearchParams({ page: String(page), pageSize: "18", withEmail: "1" });
   188→   188→    if (applied.q) sp.set("title", applied.q);
   189→   189→    if (applied.loc) sp.set("location", applied.loc);
   190→   190→    if (applied.remote) sp.set("remote", "true");
   191→   191→    return `/api/jobs?${sp.toString()}`;
   192→   192→  }, [applied, page]);
   193→   193→
   194→   194→  const { data, loading } = useApi<JobsResponse>(url);
   195→   195→  const totalPages = data ? Math.max(1, Math.ceil(data.total / data.pageSize)) : 1;
   196→   196→  const hasQuery = applied.q || applied.loc || applied.remote;
   197→   197→
   198→   198→  // ── Live scrape flow (mode "live") ──
   199→   199→  const runLiveSearch = async (q = input) => {
   200→   200→    const keyword = q.trim();
   201→   201→    if (keyword.length < 2) {
   202→   202→      toast.error("Ketik posisi yang mau dicari (min. 2 karakter)");
   203→   203→      return;
   204→   204→    }
   205→   205→    if (sources.length > 0 && selectedBoards.size === 0) {
   206→   206→      toast.error("Pilih minimal 1 board sumber dulu");
   207→   207→      return;
   208→   208→    }
   209→   209→    setLiveRunning(true);
   210→   210→    setLiveResult(null);
   211→   211→    toast.info(`Engine menjalankan scrape live ke ${selectedBoards.size} board untuk “${keyword}”…`);
   212→   212→    try {
   213→   213→      const res = await fetch("/api/search/live", {
   214→   214→        method: "POST",
   215→   215→        headers: { "Content-Type": "application/json" },
   216→   216→        body: JSON.stringify({ q: keyword, sources: [...selectedBoards] }),
   217→   217→      });
   218→   218→      const json = (await res.json()) as LiveSearchResponse & { error?: string };
   219→   219→      if (!res.ok) throw new Error(json.error ?? "Live scrape gagal");
   220→   220→      setLiveResult(json);
   221→   221→      applyNow(keyword, locInput, remote);
   222→   222→      if (json.totalCreated > 0) {
   223→   223→        toast.success(`Scrape selesai ${(json.durationMs / 1000).toFixed(1)}s — ${json.totalCreated} job baru tersimpan${json.totalSkipped > 0 ? `, ${json.totalSkipped} dilewati (tanpa email HR)` : ""}`);
   224→   224→      } else {
   225→   225→        toast.info(`Scrape selesai — ${json.totalFound} hasil, semua sudah ada di database`);
   226→   226→      }
   227→   227→    } catch (e) {
   228→   228→      toast.error(e instanceof Error ? e.message : "Live scrape gagal");
   229→   229→    } finally {
   230→   230→      setLiveRunning(false);
   231→   231→    }
   232→   232→  };
   233→   233→
   234→   234→  const pickPosition = (p: string) => {
   235→   235→    setInput(p);
   236→   236→    if (mode === "db") applyNow(p, locInput, remote);
   237→   237→    else void runLiveSearch(p);
   238→   238→  };
   239→   239→
   240→   240→  const switchMode = (m: Mode) => {
   241→   241→    setMode(m);
   242→   242→    if (m === "db") applyNow(); // instant db search with current keywords
   243→   243→  };
   244→   244→
   245→   245→  return (
   246→   246→    <div className="space-y-4">
   247→   247→      {/* Hero search */}
   248→   248→      <section className="rounded-xl border border-zinc-800 bg-gradient-to-br from-zinc-900 via-zinc-900/60 to-zinc-900/20 p-6">
   249→   249→        <div className="flex flex-wrap items-center justify-between gap-3">
   250→   250→          <div className="flex items-center gap-2.5">
   251→   251→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-amber-500/15">
   252→   252→              <BriefcaseBusiness className="h-4.5 w-4.5 text-amber-400" />
   253→   253→            </div>
   254→   254→            <div>
   255→   255→              <h2 className="text-base font-bold text-zinc-100">Cari LokerBase</h2>
   256→   256→              <p className="text-xs text-zinc-500">
   257→   257→                {mode === "db"
   258→   258→                  ? "Filter instan dari database job hasil scraping pipeline"
   259→   259→                  : "Pilih board dari Data Sources — engine scrape ngikutin setting engine di tiap source"}
   260→   260→              </p>
   261→   261→            </div>
   262→   262→          </div>
   263→   263→
   264→   264→          {/* Mode toggle */}
   265→   265→          <div className="inline-flex rounded-lg border border-zinc-800 bg-zinc-950 p-0.5">
   266→   266→            <button
   267→   267→              onClick={() => switchMode("db")}
   268→   268→              className={`inline-flex items-center gap-1.5 rounded-md px-3 py-1.5 text-xs font-medium transition-colors ${
   269→   269→                mode === "db" ? "bg-amber-500/15 text-amber-300" : "text-zinc-500 hover:text-zinc-300"
   270→   270→              }`}
   271→   271→            >
   272→   272→              <Database className="h-3.5 w-3.5" /> Cari Database
   273→   273→            </button>
   274→   274→            <button
   275→   275→              onClick={() => switchMode("live")}
   276→   276→              className={`inline-flex items-center gap-1.5 rounded-md px-3 py-1.5 text-xs font-medium transition-colors ${
   277→   277→                mode === "live" ? "bg-teal-500/15 text-teal-300" : "text-zinc-500 hover:text-zinc-300"
   278→   278→              }`}
   279→   279→            >
   280→   280→              <Zap className="h-3.5 w-3.5" /> Scrape Live (Engine)
   281→   281→            </button>
   282→   282→          </div>
   283→   283→        </div>
   284→   284→
   285→   285→        <div className="mt-4 flex flex-col gap-2 sm:flex-row">
   286→   286→          <div className="relative min-w-0 flex-1">
   287→   287→            <Search className="absolute top-1/2 left-3 h-4 w-4 -translate-y-1/2 text-zinc-500" />
   288→   288→            <Input
   289→   289→              autoFocus
   290→   290→              placeholder={mode === "db" ? "Posisi yang dicari — mis. frontend developer, data analyst…" : "Posisi buat di-scrape live — mis. business analyst…"}
   291→   291→              value={input}
   292→   292→              onChange={(e) => setInput(e.target.value)}
   293→   293→              onKeyDown={(e) => e.key === "Enter" && (mode === "db" ? applyNow() : void runLiveSearch())}
   294→   294→              disabled={liveRunning}
   295→   295→              className="h-11 border-zinc-800 bg-zinc-950 pl-9 text-sm"
   296→   296→            />
   297→   297→          </div>
   298→   298→          <div className="relative sm:w-56">
   299→   299→            <MapPin className="absolute top-1/2 left-3 h-4 w-4 -translate-y-1/2 text-zinc-500" />
   300→   300→            <Input
   301→   301→              placeholder="Lokasi (opsional)"
   302→   302→              value={locInput}
   303→   303→              onChange={(e) => setLocInput(e.target.value)}
   304→   304→              onKeyDown={(e) => e.key === "Enter" && (mode === "db" ? applyNow() : void runLiveSearch())}
   305→   305→              disabled={liveRunning}
   306→   306→              className="h-11 border-zinc-800 bg-zinc-950 pl-9 text-sm"
   307→   307→            />
   308→   308→          </div>
   309→   309→          {mode === "db" ? (
   310→   310→            <Button className="h-11 bg-amber-500 px-5 text-zinc-950 hover:bg-amber-400" onClick={() => applyNow()}>
   311→   311→              <Search className="h-4 w-4" /> Cari
   312→   312→            </Button>
   313→   313→          ) : (
   314→   314→            <Button
   315→   315→              className="h-11 bg-teal-500 px-5 text-zinc-950 hover:bg-teal-400"
   316→   316→              onClick={() => void runLiveSearch()}
   317→   317→              disabled={liveRunning}
   318→   318→            >
   319→   319→              {liveRunning ? <Loader2 className="h-4 w-4 animate-spin" /> : <Zap className="h-4 w-4" />}
   320→   320→              {liveRunning ? "Scraping…" : "Scrape Sekarang"}
   321→   321→            </Button>
   322→   322→          )}
   323→   323→        </div>
   324→   324→
   325→   325→        {/* Board picker (live mode) — sumber = Data Sources yang sudah ditambahkan */}
   326→   326→        {mode === "live" && (
   327→   327→          <div className="mt-3 flex flex-wrap items-center gap-1.5">
   328→   328→            <span className="mr-1 flex items-center gap-1 text-[11px] text-zinc-500">
   329→   329→              <Layers className="h-3.5 w-3.5" /> Board sumber:
   330→   330→            </span>
   331→   331→            {sources.map((s) => {
   332→   332→              const on = selectedBoards.has(s.slug);
   333→   333→              const pe = ENGINES[primaryEngine(s) as EngineKey];
   334→   334→              return (
   335→   335→                <button
   336→   336→                  key={s.slug}
   337→   337→                  type="button"
   338→   338→                  disabled={liveRunning}
   339→   339→                  onClick={() => toggleBoard(s.slug)}
   340→   340→                  title={`${s.name} · ${s.status} · engine: ${engineChainLabel(s)}`}
   341→   341→                  className={`inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-[11px] transition-colors disabled:opacity-50 ${
   342→   342→                    on
   343→   343→                      ? "border-teal-500/50 bg-teal-500/15 text-teal-200"
   344→   344→                      : "border-zinc-800 bg-zinc-900 text-zinc-500 hover:border-zinc-700 hover:text-zinc-300"
   345→   345→                  }`}
   346→   346→                >
   347→   347→                  <span className={`h-1.5 w-1.5 rounded-full ${STATUS_DOT[s.status] ?? "bg-zinc-600"}`} />
   348→   348→                  {s.name}
   349→   349→                  <span className={`rounded px-1 text-[9px] font-semibold ${on && pe ? pe.badge : "bg-zinc-800 text-zinc-500"}`}>
   350→   350→                    {pe?.name ?? primaryEngine(s)}
   351→   351→                  </span>
   352→   352→                </button>
   353→   353→              );
   354→   354→            })}
   355→   355→            {sources.length === 0 && (
   356→   356→              <span className="text-[11px] text-zinc-600">
   357→   357→                Belum ada source — tambahkan dulu di menu Data Sources
   358→   358→              </span>
   359→   359→            )}
   360→   360→            {sources.length > 0 && selectedBoards.size < sources.length && (
   361→   361→              <button
   362→   362→                type="button"
   363→   363→                disabled={liveRunning}
   364→   364→                onClick={() => setSelectedBoards(new Set(sources.filter((s) => s.status === "ACTIVE").map((s) => s.slug)))}
   365→   365→                className="rounded-full border border-dashed border-zinc-700 px-2.5 py-1 text-[11px] text-zinc-500 transition-colors hover:border-zinc-600 hover:text-zinc-300 disabled:opacity-50"
   366→   366→              >
   367→   367→                Reset ke semua aktif
   368→   368→              </button>
   369→   369→            )}
   370→   370→          </div>
   371→   371→        )}
   372→   372→
   373→   373→        <div className="mt-3 flex flex-wrap items-center gap-1.5">
   374→   374→          <span className="mr-1 text-[11px] text-zinc-600">{mode === "db" ? "Populer:" : "Coba scrape:"}</span>
   375→   375→          {POPULAR_POSITIONS.map((p) => (
   376→   376→            <button
   377→   377→              key={p}
   378→   378→              disabled={liveRunning}
   379→   379→              onClick={() => pickPosition(p)}
   380→   380→              className={`rounded-full border px-2.5 py-1 text-[11px] transition-colors disabled:opacity-50 ${
   381→   381→                applied.q.toLowerCase() === p.toLowerCase()
   382→   382→                  ? mode === "live"
   383→   383→                    ? "border-teal-500/50 bg-teal-500/15 text-teal-300"
   384→   384→                    : "border-amber-500/50 bg-amber-500/15 text-amber-300"
   385→   385→                  : "border-zinc-800 bg-zinc-900 text-zinc-400 hover:border-zinc-700 hover:text-zinc-200"
   386→   386→              }`}
   387→   387→            >
   388→   388→              {p}
   389→   389→            </button>
   390→   390→          ))}
   391→   391→          <label className="ml-auto flex cursor-pointer items-center gap-2 text-[11px] text-zinc-400">
   392→   392→            <Globe className="h-3.5 w-3.5 text-teal-400" /> Hanya remote
   393→   393→            <Switch checked={remote} onCheckedChange={(v) => setRemote(v)} className="data-[state=checked]:bg-teal-500" />
   394→   394→          </label>
   395→   395→        </div>
   396→   396→
   397→   397→        {mode === "live" && liveRunning && (
   398→   398→          <p className="mt-3 flex items-center gap-2 rounded-lg border border-teal-500/20 bg-teal-500/5 px-3 py-2 text-xs text-teal-300">
   399→   399→            <Loader2 className="h-3.5 w-3.5 animate-spin" />
   400→   400→            Engine menjalankan scrape live ke {selectedNames.length > 0 ? selectedNames.join(", ") : "board aktif"} — engine ngikutin setting tiap source · hanya lowongan dengan email HR yang disimpan…
   401→   401→          </p>
   402→   402→        )}
   403→   403→
   404→   404→        {/* Live scrape breakdown per board */}
   405→   405→        {mode === "live" && liveResult && (
   406→   406→          <div className="mt-3 space-y-2">
   407→   407→            <p className="text-xs text-zinc-500">
   408→   408→              Selesai dalam <span className="font-semibold tabular-nums text-zinc-300">{(liveResult.durationMs / 1000).toFixed(1)}s</span> —{" "}
   409→   409→              <span className="font-semibold text-teal-300">{liveResult.totalCreated} job baru</span> tersimpan,{" "}
   410→   410→              {liveResult.totalDuplicate} duplikat dilewati{" "}
   411→   411→              {liveResult.totalSkipped > 0 && (
   412→   412→                <>
   413→   413→                  · <span className="text-zinc-400">{liveResult.totalSkipped} dilewati (tanpa email HR)</span>{" "}
   414→   414→                </>
